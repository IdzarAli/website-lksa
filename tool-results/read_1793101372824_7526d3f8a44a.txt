   172→function FormulirLogin({ padaSukses, kembali }: { padaSukses: (s: Sesi) => void; kembali: () => void }) {
   173→  const [email, setEmail] = useState("");
   174→  const [password, setPassword] = useState("");
   175→  const [galat, setGalat] = useState("");
   176→  const [memuat, setMemuat] = useState(false);
   177→
   178→  const kirim = async (e: React.FormEvent) => {
   179→    e.preventDefault();
   180→    setMemuat(true);
   181→    setGalat("");
   182→    try {
   183→      const res = await fetch("/api/admin/login", {
   184→        method: "POST",
   185→        headers: { "Content-Type": "application/json" },
   186→        body: JSON.stringify({ email, password }),
   187→      });
   188→      const d = await res.json();
   189→      if (!res.ok) {
   190→        setGalat(d.error ?? "Gagal masuk");
   191→        return;
   192→      }
   193→      padaSukses({ nama: d.nama, role: d.role, email });
   194→    } catch {
   195→      setGalat("Koneksi gagal");
   196→    } finally {
   197→      setMemuat(false);
   198→    }
   199→  };
   200→
   201→  return (
   202→    <main id="konten" className="mx-auto max-w-md px-4 py-16 sm:px-6">
   203→      <div className="rounded-2xl border border-brand-100 bg-white p-6">
   204→        <h1 className="flex items-center gap-2 text-2xl">
   205→          <Lock className="h-5 w-5 text-brand-600" aria-hidden />
   206→          Masuk Ruang Pengurus
   207→        </h1>
   208→        <p className="mt-2 text-sm text-muted-foreground">
   209→          Untuk petugas lembaga. Peran menentukan menu yang tersedia.
   210→        </p>
   211→        {galat && (
   212→          <p className="mt-3 rounded-lg border border-cta-100 bg-cta-50 p-3 text-sm text-cta-700" role="alert">{galat}</p>
   213→        )}
   214→        <form className="mt-4 space-y-4" onSubmit={kirim}>
   215→          <div>
   216→            <label htmlFor="adm-email" className="mb-1 block text-sm font-semibold">Email</label>
   217→            <input id="adm-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} required className="w-full rounded-lg border border-input px-3 py-2.5 text-sm" autoComplete="username" />
   218→          </div>
   219→          <div>
   220→            <label htmlFor="adm-sandi" className="mb-1 block text-sm font-semibold">Password</label>
   221→            <input id="adm-sandi" type="password" value={password} onChange={(e) => setPassword(e.target.value)} required className="w-full rounded-lg border border-input px-3 py-2.5 text-sm" autoComplete="current-password" />
   222→          </div>
   223→          <button type="submit" disabled={memuat} className="min-h-[44px] w-full rounded-lg bg-brand-600 px-4 py-3 text-sm font-semibold text-white hover:bg-brand-700 disabled:opacity-60">
   224→            {memuat ? "Memeriksa…" : "Masuk"}
   225→          </button>
   226→        </form>
   227→        <div className="mt-4 rounded-lg border border-amber-300 bg-amber-50 p-3 text-xs leading-relaxed text-amber-900">
   228→          <strong>Akun demo (CONTOH — ganti sebelum produksi):</strong>
   229→          <br />editor@contoh.id / editor123 · verifikator@contoh.id / verifikator123 · pengelola@contoh.id / pengelola123
   230→        </div>
   231→        <button type="button" onClick={kembali} className="mt-4 text-sm font-semibold text-brand-700 underline-offset-4 hover:underline">
   232→          ← Kembali ke situs publik
   233→        </button>
   234→      </div>
   235→    </main>
   236→  );
   237→}

   239→/* ================= ANTREAN DONASI ================= */
   240→function AntreanDonasi() {
   241→  const [items, setItems] = useState<Donasi[] | null>(null);
   242→  const [filter, setFilter] = useState("");
   243→  const [versi, setVersi] = useState(0);
   244→  const [buka, setBuka] = useState<Donasi | null>(null);
   245→
   246→  useEffect(() => {
   247→    let aktif = true;
   248→    fetch(`/api/admin/donations${filter ? `?status=${filter}` : ""}`)
   249→      .then((r) => (r.ok ? r.json() : { items: [] }))
   250→      .then((d) => {
   251→        if (aktif) setItems(d.items ?? []);
   252→      })
   253→      .catch(() => {
   254→        if (aktif) setItems([]);
   255→      });
   256→    return () => {
   257→      aktif = false;
   258→    };
   259→  }, [filter, versi]);
   260→
   261→  const muatData = () => setVersi((v) => v + 1);
   262→
   263→  return (
   264→    <div>
   265→      <div className="flex flex-wrap items-center justify-between gap-3">
   266→        <div>
   267→          <h2 className="text-xl">Antrean konfirmasi donasi</h2>
   268→          <p className="text-sm text-muted-foreground">
   269→            Hanya petugas berperan verifikator yang dapat mengubah status; editor ditolak server.
   270→          </p>
   271→        </div>
   272→        <div className="flex items-center gap-2">
   273→          <label htmlFor="filter-donasi" className="text-sm font-semibold">Status</label>
   274→          <select id="filter-donasi" value={filter} onChange={(e) => setFilter(e.target.value)} className="rounded-lg border border-input px-3 py-2.5 text-sm">
   275→            <option value="">Semua</option>
   276→            <option value="menunggu-verifikasi">Menunggu verifikasi</option>
   277→            <option value="terverifikasi">Terverifikasi</option>
   278→            <option value="perlu-perbaikan">Perlu perbaikan</option>
   279→            <option value="ditolak">Ditolak</option>
   280→          </select>

   282→            <RefreshCcw className="h-4 w-4" aria-hidden /> Muat ulang
   283→          </button>
   284→        </div>
   285→      </div>
   286→
   287→      <div className="mt-4 max-h-[32rem] overflow-y-auto scroll-area-thin rounded-xl border border-brand-100">
   288→        <table className="w-full text-sm">
   289→          <thead className="sticky top-0 bg-brand-50 text-left">
   290→            <tr>
   291→              <th scope="col" className="px-3 py-2.5 font-semibold">Referensi</th>
   292→              <th scope="col" className="px-3 py-2.5 font-semibold">Donatur</th>
   293→              <th scope="col" className="px-3 py-2.5 font-semibold">Nominal</th>
   294→              <th scope="col" className="px-3 py-2.5 font-semibold">Status</th>
   295→              <th scope="col" className="px-3 py-2.5 font-semibold">Aksi</th>
   296→            </tr>
   297→          </thead>
   298→          <tbody>
   299→            {items === null && (
   300→              <tr><td colSpan={5} className="px-3 py-6 text-center text-muted-foreground">Memuat…</td></tr>
   301→            )}
   302→            {items !== null && items.length === 0 && (
   303→              <tr><td colSpan={5} className="px-3 py-6 text-center text-muted-foreground">Tidak ada data.</td></tr>
   304→            )}

   306→              <tr key={d.id} className="border-t border-border/60">

   310→                <td className="px-3 py-2.5"><StatusBadge status={d.status} /></td>
   311→                <td className="px-3 py-2.5">
   312→                  <button type="button" onClick={() => setBuka(d)} className="rounded-lg border border-brand-300 px-3 py-1.5 text-xs font-semibold text-brand-700 hover:bg-brand-50">
   313→                    Buka
   314→                  </button>
   315→                </td>
   316→              </tr>
   317→            ))}
   318→          </tbody>
   319→        </table>
   320→      </div>
   321→
   322→      {buka && <DetailDonasi donasi={buka} tutup={() => setBuka(null)} padaBerubah={muatData} />}
   323→    </div>

   327→function DetailDonasi({ donasi, tutup, padaBerubah }: { donasi: Donasi; tutup: () => void; padaBerubah: () => void }) {
   328→  const [buktiAda, setBuktiAda] = useState<boolean | null>(null);
   329→  const [statusBaru, setStatusBaru] = useState("");
   330→  const [alasan, setAlasan] = useState("");
   331→  const [galat, setGalat] = useState("");
   332→  const [memuat, setMemuat] = useState(false);
   333→
   334→  const opsi = OPSI_STATUS_DONASI[donasi.status] ?? [];
   335→
   336→  const keputusan = async () => {
   337→    setGalat("");
   338→    if (!statusBaru) { setGalat("Pilih status tujuan"); return; }
   339→    if (alasan.trim().length < 5) { setGalat("Alasan keputusan wajib diisi (minimal 5 karakter)"); return; }
   340→    setMemuat(true);
   341→    try {
   342→      const res = await fetch(`/api/admin/donations/${donasi.id}/status`, {
   343→        method: "PATCH",
   344→        headers: { "Content-Type": "application/json" },
   345→        body: JSON.stringify({ status: statusBaru, alasan: alasan.trim(), version: donasi.version }),
   346→      });
   347→      const d = await res.json();
   348→      if (!res.ok) {
   349→        setGalat(d.errors?.version ?? d.errors?.alasan ?? d.error ?? "Gagal mengubah status");
   350→        return;
   351→      }
   352→      tutup();
   353→      padaBerubah();
   354→    } finally {
   355→      setMemuat(false);
   356→    }
   357→  };
   358→
   359→  return (
   360→    <div className="fixed inset-0 z-50 flex items-center justify-center bg-black/40 p-4" role="dialog" aria-modal="true" aria-label={`Detail donasi ${donasi.referensi}`}>
   361→      <div className="max-h-[85vh] w-full max-w-2xl overflow-y-auto scroll-area-thin rounded-2xl bg-white p-6">
   362→        <div className="flex items-start justify-between gap-3">
   363→          <div>
   364→            <h3 className="font-mono text-lg font-bold">{donasi.referensi}</h3>
   365→            <StatusBadge status={donasi.status} />
   366→          </div>
   367→          <button type="button" onClick={tutup} className="rounded-lg border border-border px-3 py-2 text-sm font-semibold hover:bg-muted" aria-label="Tutup detail">Tutup</button>
   368→        </div>
   369→
   370→        <dl className="mt-4 grid gap-x-6 gap-y-2 text-sm sm:grid-cols-2">
   371→          <div><dt className="text-muted-foreground">Donatur</dt><dd className="font-semibold">{donasi.nama}</dd></div>
   372→          <div><dt className="text-muted-foreground">Kontak</dt><dd>{donasi.email} · {donasi.wa}</dd></div>
   373→          <div><dt className="text-muted-foreground">Nominal</dt><dd className="font-semibold tabular-nums">{formatRupiah(donasi.nominalRupiah)}</dd></div>
   374→          <div><dt className="text-muted-foreground">Keterangan</dt><dd>{donasi.keterangan || "—"}</dd></div>
   375→          <div><dt className="text-muted-foreground">Persetujuan leaderboard</dt><dd>{donasi.leaderboardConsent ? "diminta" : "tidak"}</dd></div>
   376→          <div><dt className="text-muted-foreground">Diterima</dt><dd>{formatTanggal(donasi.createdAt)}</dd></div>
   377→          {donasi.verifiedBy && (
   378→            <>
   379→              <div><dt className="text-muted-foreground">Diputuskan oleh</dt><dd>{donasi.verifiedBy} · {formatTanggal(donasi.verifiedAt)}</dd></div>
   380→              <div><dt className="text-muted-foreground">Alasan</dt><dd>{donasi.keputusanAlasan}</dd></div>
   381→            </>
   382→          )}
   383→        </dl>
   384→
   385→        {/* bukti HANYA lewat route terautentikasi (INV-4) */}
   386→        <div className="mt-4 rounded-xl border border-brand-100 p-4">
   387→          <h4 className="flex items-center gap-2 text-sm font-semibold">
   388→            <ImageIcon className="h-4 w-4 text-brand-600" aria-hidden />
   389→            Bukti transfer (privat)
   390→          </h4>
   391→          <p className="mt-1 text-xs text-muted-foreground">
   392→            Kunci berkas: <span className="font-mono">{donasi.receiptKey.slice(0, 18)}…</span> — hanya terbuka lewat route berperan verifikator.
   393→          </p>
   394→          {buktiAda === null && (
   395→            <button
   396→              type="button"
   397→              onClick={() => {
   398→                fetch(`/api/admin/receipts/${donasi.receiptKey}`)
   399→                  .then((r) => setBuktiAda(r.ok))
   400→                  .catch(() => setBuktiAda(false));
   401→              }}
   402→              className="mt-2 inline-flex min-h-[44px] items-center gap-2 rounded-lg bg-brand-600 px-4 py-2.5 text-sm font-semibold text-white hover:bg-brand-700"
   403→            >
   404→              <Download className="h-4 w-4" aria-hidden /> Buka bukti
   405→            </button>
   406→          )}
